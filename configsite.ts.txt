  // components/ui/DownloadForm.tsx
  "use client";
  import React, { useState } from 'react';

  type Status = 'idle' | 'loading' | 'success' | 'error';

  export default function DownloadForm() {
    const [url, setUrl] = useState('');
    const [status, setStatus] = useState<Status>('idle');
    const [result, setResult] = useState<{ title: string; quality: string; size: string } | null>(null);

    const handleDownload = async (e: React.FormEvent) => {
      e.preventDefault();
      if (!url) return;

      setStatus('loading');

      try {
        // Simulation d'appel API
        const response = await fetch('/api/download', {
          method: 'POST',
          body: JSON.stringify({ url }),
          headers: { 'Content-Type': 'application/json' },
        });

        if (!response.ok) throw new Error('Lien invalide');

        const data = await response.json();
        setResult(data);
        setStatus('success');
      } catch (err) {
        setStatus('error');
      }
    };

    return (
      <div className="w-full max-w-3xl mx-auto">
        <form onSubmit={handleDownload} className="relative flex flex-col sm:flex-row gap-3 p-2 bg-white rounded-2xl
  shadow-premium border border-gray-100">
          <div className="relative flex-1">
            <input
              type="url"
              placeholder="Collez le lien de votre vidéo ici..."
              className="w-full px-4 py-4 rounded-xl bg-transparent outline-none text-accent-dark
  placeholder:text-gray-400"
              value={url}
              onChange={(e) => setUrl(e.target.value)}
              required
            />
          </div>
          <button
            type="submit"
            disabled={status === 'loading'}
            className="btn-primary whitespace-nowrap flex items-center justify-center gap-2"
          >
            {status === 'loading' ? (
              <>
                <svg className="animate-spin h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none"
  viewBox="0 0 24 24"><circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor"
  strokeWidth="4"></circle><path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0
  12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.918l3-2.627z"></path></svg>
                Analyse...
              </>
            ) : 'Télécharger'}
          </button>
        </form>

        {/* Feedback Area */}
        <div className="mt-6 transition-all duration-300">
          {status === 'error' && (
            <div className="p-4 bg-red-50 border border-red-100 text-red-600 rounded-xl text-center text-sm animate-in
  fade-in slide-in-from-top-2">
              Ce lien ne peut pas être traité. Vérifiez l'URL et assurez-vous que vous disposez des droits nécessaires
  pour télécharger ce contenu.
            </div>
          )}

          {status === 'success' && result && (
            <div className="p-6 bg-white border border-green-100 rounded-2xl shadow-soft animate-in zoom-in-95
  duration-300">
              <div className="flex items-center gap-4">
                <div className="bg-green-100 p-3 rounded-full text-green-600">
                  <svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none"
  stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><polyline points="20 6 9 17 4
  12"/></svg>
                </div>
                <div className="flex-1">
                  <h3 className="font-bold text-accent-dark">{result.title}</h3>
                  <p className="text-sm text-accent-medium">{result.quality} • {result.size}</p>
                </div>
                <a href="#" className="btn-primary !py-2 !px-4 text-sm">Lancer le téléchargement</a>
              </div>
            </div>
          )}
        </div>
      </div>
    );
  }

  import { NextResponse } from 'next/server';

  // Liste des domaines autorisés pour éviter les attaques SSRF (Server-Side Request Forgery)
  const ALLOWED_DOMAINS = [
    'youtube.com', 'youtu.be', 'tiktok.com', 'facebook.com', 'instagram.com', 'fb.watch'
  ];

  export async function POST(request: Request) {
    try {
      const body = await request.json();
      const { url } = body;

      // 1. Validation de l'URL
      if (!url || !url.startsWith('http')) {
        return NextResponse.json({ error: 'URL invalide' }, { status: 400 });
      }

      try {
        const parsedUrl = new URL(url);
        const domain = parsedUrl.hostname.replace('www.', '');

        // 2. Vérification du domaine
        const isAllowed = ALLOWED_DOMAINS.some(d => domain.endsWith(d));
        if (!isAllowed) {
          return NextResponse.json({
            error: 'Cette plateforme n\'est pas encore supportée ou le lien est invalide.'
          }, { status: 400 });
        }

        // 3. Simulation du traitement (C'est ici que vous connecterez yt-dlp ou une API tierce)
        // console.log(`Traitement de la vidéo : ${url}`);

        // Simulation d'un délai de réseau
        await new Promise(resolve => setTimeout(resolve, 1500));

        // 4. Retour des informations de la vidéo
        // Dans un cas réel, ces données proviendraient de l'analyse du lien
        return NextResponse.json({
          title: "Vidéo téléchargée avec succès",
          quality: "1080p (HD)",
          size: "12.4 MB",
          downloadUrl: "#", // URL générée par votre service de téléchargement
        });

      } catch (e) {
        return NextResponse.json({ error: 'Format d\'URL incorrect' }, { status: 400 });
      }

    } catch (error) {
      return NextResponse.json({ error: 'Erreur interne du serveur' }, { status: 500 });
    }
  }
